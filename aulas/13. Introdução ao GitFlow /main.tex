\section{Fundamentos: O que é o Git?}

% Slide Git 1
\begin{frame}{O que é o Git?}
    \begin{itemize}
        \item O Git é um \textbf{Sistema de Controle de Versão Distribuído} (DVCS).
        \item Ferramenta essencial para o desenvolvimento de software moderno.
        \item Permite rastrear mudanças em arquivos e coordenar o trabalho entre várias pessoas.
        \item É de código aberto (\textit{Open Source}) e totalmente gratuito.
    \end{itemize}
\end{frame}

% Slide Git 2
\begin{frame}{Origem do Git}
    \begin{itemize}
        \item Criado em \textbf{2005} por \textbf{Linus Torvalds} (o mesmo criador do Linux).
        \item Motivação: A comunidade de desenvolvimento do Kernel Linux precisava de uma ferramenta rápida, descentralizada e que suportasse fluxos de trabalho não lineares (milhares de desenvolvedores independentes).
        \item "Git" é uma gíria britânica para "pessoa teimosa ou ignorante" (Linus brincou que sempre nomeia os projetos com seu próprio nome).
    \end{itemize}
\end{frame}

% Slide Git 3
\begin{frame}{Por que usar Controle de Versão?}
    Sem controle de versão, lidamos com arquivos assim:
    \begin{center}
        \texttt{projeto\_final.doc} \\
        \texttt{projeto\_final\_v2.doc} \\
        \texttt{projeto\_final\_agora\_vai.doc} \\
        \texttt{projeto\_final\_sergio\_revisado.doc}
    \end{center}
    \vspace{0.3cm}
    \textbf{Com o Git:}
    \begin{itemize}
        \item Todo o histórico fica encapsulado na pasta do projeto.
        \item Se o projeto quebrar, é possível voltar no tempo para qualquer versão anterior.
        \item Fica registrado \textbf{quem} modificou o quê, \textbf{quando} e \textbf{por quê}.
    \end{itemize}
\end{frame}

% Slide Git 4
\begin{frame}{Centralizado vs Distribuído}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Sistemas Centralizados (SVN, CVS)}
            \begin{itemize}
                \item Existe apenas \textit{um} servidor central com o histórico.
                \item Se o servidor cair, ninguém consegue versionar o código.
                \item Operações dependem de internet.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Sistemas Distribuídos (Git)}
            \begin{itemize}
                \item Cada desenvolvedor tem um "clone" completo do histórico na própria máquina.
                \item Se o servidor explodir, qualquer máquina local serve como backup total.
                \item Maioria das operações é local e instantânea.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide Git 5
\begin{frame}{Como o Git pensa: Snapshots}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Outros sistemas (SVN) armazenam alterações baseadas em arquivo (\textbf{deltas}).
                \item O Git pensa em seus dados como uma série de "fotografias" (\textbf{snapshots}).
                \item Ao salvar o estado, o Git tira uma foto do sistema de arquivos e armazena uma referência.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[x=1cm, y=0.8cm, every node/.style={font=\scriptsize}]
                    \node[draw, rounded corners, fill=blue!10] (v1) at (0,2) {Versão 1};

                    \draw[->, thick, dashed] (v1) -- node[right] {\tiny $\Delta$ (delta)} (v2);
                    \node[above=0.2cm of v1] {\textbf{SVN (Deltas)}};
                    
                    \node[draw, rounded corners, fill=green!10] (s1) at (3,2) {Snapshot A};

                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% Slide Git 6
\begin{frame}{Os Três Estados do Git}
    \begin{columns}
        \begin{column}{0.6\textwidth}
            Arquivos no Git podem residir em três estados:
            \vspace{0.3cm}
            \begin{enumerate}
                \item \textbf{Modificado:} O arquivo foi alterado, mas ainda não foi salvo.
                \item \textbf{Preparado (\textit{Staged}):} Marcado para ir no próximo commit.
                \item \textbf{Consolidado (\textit{Committed}):} Armazenado de forma segura.
            \end{enumerate}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{center}
                \begin{tikzpicture}[node distance=1.5cm, every node/.style={draw, rounded corners, align=center, font=\scriptsize}]
                    \node[fill=blue!10] (mod) {Modificado};
                    \node[fill=orange!10, below of=mod] (stage) {Preparado};
                    \node[fill=green!10, below of=stage] (com) {Consolidado};
                    
                    \draw[->, thick] (mod) -- node[right, draw=none, font=\tiny] {\texttt{git add}} (stage);
                    \draw[->, thick] (stage) -- node[right, draw=none, font=\tiny] {\texttt{git commit}} (com);
                    \draw[->, thick, dashed] (com) to[bend left=45] node[left, draw=none, font=\tiny] {edita} (mod);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% Slide Git 8
\begin{frame}{Por que o Git dominou o mercado?}
    \begin{itemize}
        \item \textbf{Velocidade:} Operações locais não dependem de rede.
        \item \textbf{Design simples:} Embora poderoso, a estrutura de dados interna é elegante.
        \item \textbf{Suporte forte a branches:} Criar e mesclar branches no Git leva frações de segundos, o que encoraja fluxos de trabalho baseados em ramificações (como o próprio GitFlow).
        \item \textbf{Integridade Criptográfica:} Tudo no Git gera um hash SHA-1. É impossível alterar um arquivo, data ou autor sem que o Git saiba.
    \end{itemize}
\end{frame}

\section{Git vs GitHub}

% Slide 11
\begin{frame}{O que é o GitHub?}
    \begin{itemize}
        \item O GitHub é uma \textbf{plataforma de hospedagem} de código-fonte em nuvem que utiliza o Git.
        \item Adiciona funcionalidades de rede social ao desenvolvimento (seguir projetos, estrelas, issues, pull requests).
        \item Foi comprado pela Microsoft em 2018 e é o maior repositório de código do planeta.
        \item Existem alternativas concorrentes que também usam Git: \textbf{GitLab}, \textbf{Bitbucket}, \textbf{Gitea}.
    \end{itemize}
\end{frame}

% Slide 12
\begin{frame}{Diferença Fundamental: Git $\times$ GitHub}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \textbf{\Huge Git}
            \end{center}
            \vspace{0.2cm}
            \begin{itemize}
                \item É a \textbf{ferramenta}.
                \item Software de linha de comando instalado no seu computador.
                \item Trabalha localmente e \textit{offline}.
                \item Não possui interface gráfica nativa (apenas terminal).
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \textbf{\Huge GitHub}
            \end{center}
            \vspace{0.2cm}
            \begin{itemize}
                \item É o \textbf{serviço} web.
                \item Site na internet hospedando repositórios Git.
                \item Trabalha na nuvem (\textit{online}).
                \item Oferece interface gráfica rica, gestão de projetos, CI/CD.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Introdução ao GitFlow e Problemas}

% Slide 1
\begin{frame}[fragile]{Problemas no Desenvolvimento Colaborativo (O Caos)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            Antes de adotar um fluxo estruturado, as equipes sofrem com:
            \vspace{0.2cm}
            \begin{itemize}
                \item \textbf{Código Quebrado:} Funcionalidades inacabadas vão parar no cliente acidentalmente.
                \item \textbf{Conflitos Constantes:} Desenvolvedores alterando os mesmos arquivos na \texttt{main} ao mesmo tempo.
                \item \textbf{Sem Rastreabilidade:} "Quem subiu esse código? Qual versão corrigiu aquele problema?"
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}

\end{frame}

\section{Conceito de Git Flow}

% Slide 3
\begin{frame}{Conceito de Git Flow}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Criado por \textbf{Vincent Driessen} (2010).
                \item Não é um comando nativo ou plugin obrigatório, mas sim um \textbf{conjunto estrito de regras} e convenções.
                \item \textbf{Objetivo:} Fornecer um modelo de ramificação rigoroso para organizar projetos em larga escala.
                \item Excelente para produtos de software tradicionais com \textbf{cronogramas de lançamento} definidos.
            \end{itemize}

\section{Vantagens e Desvantagens}

% Slide 5
\begin{frame}{Vantagens e Desvantagens}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{Vantagens \textcolor{meuverde}{\textbf{(+)}}}
                \begin{itemize}
                    \item Histórico de projeto limpo e rastreável.
                    \item Separação clara entre desenvolvimento, testes e produção.
                    \item Facilita o trabalho paralelo em grandes equipes.
                    \item Suporta múltiplas versões ativas do software simultaneamente.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Desvantagens \textcolor{meuvermelhoescuro}{\textbf{(-)}}}
                \begin{itemize}
                    \item Histórico denso com muitos \textit{merge commits}.
                    \item Curva de aprendizado moderada para novos desenvolvedores.
                    \item Complexidade desnecessária para projetos pequenos ou startups.
                    \item Pode conflitar com estratégias de Entrega Contínua (Continuous Deployment).
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Unidade 2 -- Estrutura do Git Flow}

% Slide 6
\begin{frame}{Unidade 2 -- Estrutura do Git Flow}
    A estrutura do Git Flow categoriza os branches com base em seu \textbf{tempo de vida} e \textbf{propósito}.
    
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Branches Principais (Longa duração):}
        \begin{itemize}
            \item \texttt{main} (ou \texttt{master}): Código em produção, altamente estável.
            \item \texttt{develop}: Base para o próximo release, onde o código é integrado.
            \item \texttt{staging}: (Opcional, mas comum) Ambiente de homologação/pré-produção.
        \end{itemize}
        \vspace{0.3cm}
        \item \textbf{Branches Auxiliares (Curta duração):}
        \begin{itemize}
            \item \texttt{feature/*}: Desenvolvimento de novas funcionalidades.
            \item \texttt{release/*}: Preparação de uma versão para lançamento.
            \item \texttt{hotfix/*}: Correções de emergência em produção.
        \end{itemize}
    \end{itemize}
\end{frame}

% Slide 7
\begin{frame}[fragile]{Branches Principais: \texttt{main} e \texttt{develop}}
    \begin{itemize}
        \item A \texttt{main} só recebe merges de \texttt{release} ou \texttt{hotfix}. Cada commit na \texttt{main} geralmente recebe uma Tag (ex: \texttt{v1.2.0}).
        \item A \texttt{develop} contém as últimas alterações entregues para a próxima versão.
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Exemplo de inicialização:}
\begin{lstlisting}
# Inicializa o repositorio
git init

# O primeiro commit vai para a main
git commit --allow-empty -m "Commit inicial"

# Criando a branch develop a partir da main
git branch develop
git checkout develop
\end{lstlisting}
\end{frame}

% Slide 7.5
\begin{frame}[fragile]{Iniciando o GitFlow automaticamente (\texttt{git flow init})}
    \begin{itemize}
        \item O comando \textbf{\texttt{git flow init}} inicializa um repositório Git em uma pasta e configura toda a estrutura de ramificações automaticamente.
        \item Ele fará uma série de perguntas sobre como nomear as ramificações. O padrão é pressionar \texttt{Enter} para todas.
    \end{itemize}
    \vspace{0.3cm}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
$ git flow init
No branches exist yet. Base branches must be created now.
Branch name for production releases: [main] 
Branch name for "next release" development: [develop] 

How to name your supporting branch prefixes?
Feature branches? [feature/] 
Release branches? [release/] 
Hotfix branches? [hotfix/] 
\end{lstlisting}
\end{frame}

% Slide 8
\begin{frame}[fragile]{A Branch \texttt{staging} (Homologação)}
    \begin{itemize}
        \item Embora o modelo original de Vincent Driessen não a cite explicitamente, a branch \texttt{staging} é amplamente usada em cenários corporativos reais.
        \item Atua como uma ponte entre a \texttt{develop}/\texttt{release} e a \texttt{main}.
        \item \textbf{Propósito:} Implantar o código em um ambiente espelho da produção para testes finais de QA (Quality Assurance) e aceite do cliente.
    \end{itemize}
    \vspace{0.3cm}
\begin{lstlisting}
# Levando o codigo da develop para staging para testes
git checkout staging
git merge develop --no-ff
git push origin staging
\end{lstlisting}
\end{frame}

% Slide 9
\begin{frame}[fragile]{Branches Auxiliares: \texttt{feature}}
    \begin{itemize}
        \item \textbf{Origem:} \texttt{develop} \quad | \quad \textbf{Destino:} \texttt{develop}
        \item Usadas para desenvolver novas funcionalidades.
    \end{itemize}
    
    \begin{columns}
        \begin{column}{0.65\textwidth}
\begin{lstlisting}
# Criando a feature
git checkout -b feature/login develop

# Trabalhando e "commitando"
git commit -am "Adiciona tela de login"

# Finalizando a feature (merge em develop)
git checkout develop
git merge --no-ff feature/login
git branch -d feature/login
\end{lstlisting}
        \end{column}
        \begin{column}{0.35\textwidth}
            \begin{center}
            \begin{tikzpicture}[node distance=1cm]
                \node[circle, fill=orange!50, inner sep=2pt] (d1) {};
                \node[circle, fill=green!50, inner sep=2pt, above right=0.3cm and 0.5cm of d1] (f1) {};
                \node[circle, fill=orange!50, inner sep=2pt, below right=0.3cm and 0.5cm of f1] (d2) {};
                \draw[->, thick, orange] (d1) -- (d2);

                \draw[->, thick, green] (f1) to[bend left] (d2);

                \node[above=0.1cm of f1, font=\tiny] {\texttt{feature/X}};
            \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% Slide 10
\begin{frame}[fragile]{Branches Auxiliares: \texttt{release} e \texttt{hotfix}}
    \begin{itemize}
        \item \textbf{\texttt{release/*}:} (Origem: \texttt{develop}) Para estabilizar a próxima versão (apenas \textit{bugfixes}). Finaliza em \texttt{main} e \texttt{develop}.
        \item \textbf{\texttt{hotfix/*}:} (Origem: \texttt{main}) Correção de emergência em produção. Finaliza em \texttt{main} e \texttt{develop}.
    \end{itemize}
    \vspace{0.2cm}
\begin{lstlisting}
# Criando um hotfix a partir da main
git checkout -b hotfix/v1.0.1 main
git commit -am "Corrige bug critico"

# Finalizando (merge em main e develop)
git checkout main
git merge --no-ff hotfix/v1.0.1
git tag -a v1.0.1 -m "Versao 1.0.1"

git checkout develop
git merge --no-ff hotfix/v1.0.1
\end{lstlisting}
\end{frame}

\section{Fluxo de Trabalho Detalhado}

% Slide 11 (Slide 1 do Fluxo)
\begin{frame}{1. Ciclo de Vida no GitFlow}
    O \textbf{fluxo de trabalho} do GitFlow descreve o ciclo de vida de uma alteração no código, desde a concepção até a chegada em produção.
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Desenvolvimento de funcionalidades} (\textit{features}).
        \item \textbf{Integração do código} (\textit{merge} na \texttt{develop}).
        \item \textbf{Preparação de versões} (\textit{releases}).
        \item \textbf{Correções emergenciais} (\textit{hotfixes}).
        \item \textbf{Versionamento} (\textit{tags} semânticas).
    \end{enumerate}
\end{frame}

% Slide 12 (Slide 2 do Fluxo)
\begin{frame}{2. Desenvolvimento de Funcionalidades}
    \begin{itemize}
        \item Toda nova ideia, requisito ou alteração técnica começa com uma nova branch de \textbf{feature}.
        \item \textbf{Regra de ouro:} \textbf{NUNCA} desenvolva ou comite diretamente na \texttt{develop} ou na \texttt{main}.
        \item O isolamento permite que dezenas de desenvolvedores trabalhem em tarefas diferentes sem gerar conflitos diários ou quebrar o código dos colegas.
    \end{itemize}
\end{frame}

% Slide 13 (Slide 3 do Fluxo)
\begin{frame}[fragile]{3. Criando uma Feature Localmente}
    A branch de feature \textbf{sempre} nasce a partir da \texttt{develop}.
    \vspace{0.3cm}
\begin{lstlisting}
# 1. Garanta que voce esta na develop
git checkout develop

# 2. Puxe as ultimas alteracoes do servidor (boa pratica)
git pull origin develop

# 3. Crie e mude para a nova branch de feature
# O prefixo 'feature/' ajuda na organizacao visual
git checkout -b feature/relatorio-vendas
\end{lstlisting}
\end{frame}

% Slide 14 (Slide 4 do Fluxo)
\begin{frame}{4. Trabalhando na Feature}
    \textbf{Boas Práticas durante o desenvolvimento da Feature:}
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Commits pequenos e atômicos:} Faça commits lógicos (ex: "Cria botão X", "Adiciona rota Y") em vez de um gigantesco no final do dia.
        \item \textbf{Mensagens claras:} A mensagem do commit deve explicar o \textit{porquê} da mudança.
        \item \textbf{Sincronização:} Se o projeto for muito ativo e a \texttt{develop} mudar muito enquanto você desenvolve, faça \textit{merge} (ou \textit{rebase}) da \texttt{develop} para dentro da sua feature frequentemente.
    \end{itemize}
\end{frame}

% Slide 16 (Slide 6 do Fluxo)
\begin{frame}{6. Integração na Branch \texttt{develop}}
    \begin{itemize}
        \item Quando a feature está pronta, testada e completa, ela precisa voltar para o código comum (\texttt{develop}).
        \item Essa integração (\textit{merge}) precisa ser feita com cuidado para evitar quebrar a compilação global do projeto.
        \item A abordagem de integração depende se a equipe usa revisão de código centralizada ou trabalho local puramente colaborativo.
    \end{itemize}
\end{frame}

% Slide 17 (Slide 7 do Fluxo)
\begin{frame}{7. Pull Requests (PR) vs Merge Local}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{Pull Request (Github/Gitlab)}
                \begin{itemize}
                    \item A feature é empurrada (\textit{push}) para o servidor remoto.
                    \item O desenvolvedor abre uma "Solicitação de Integração".
                    \item Outros membros revisam o código (\textit{Code Review}).
                    \item A plataforma realiza o merge automaticamente.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Merge Local (CLI)}
                \begin{itemize}
                    \item Usado em equipes pequenas ou repositórios locais.
                    \item O próprio desenvolvedor faz o \textit{merge} na sua máquina.
                    \item A equipe confia totalmente nos testes e na palavra do desenvolvedor que está integrando.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% Slide 18 (Slide 8 do Fluxo)
\begin{frame}[fragile]{8. Merge Local da Feature}
    Se a equipe optar pelo merge via terminal local:
    \vspace{0.3cm}
\begin{lstlisting}
# 1. Volte para a develop
git checkout develop

# 2. Atualize com o servidor (evita conflitos pre-merge)
git pull origin develop

# 3. Faca o merge usando a flag --no-ff obrigatoriamente
git merge --no-ff feature/relatorio-vendas

# 4. Apague a branch de feature (limpeza do historico)
git branch -d feature/relatorio-vendas
\end{lstlisting}
\end{frame}

% Slide 20 (Slide 10 do Fluxo)
\begin{frame}{10. Preparação de Releases}
    \begin{itemize}
        \item Quando a \texttt{develop} atinge um estado estável contendo as features planejadas, cria-se uma branch de \textbf{release}.
        \item \textbf{Congelamento de Features:} A branch de \texttt{release} sofre um congelamento parcial. \textbf{Não são mais aceitas novas funcionalidades!}
        \item O que é permitido na branch \texttt{release}?
        \begin{itemize}
            \item Correção de pequenos bugs encontrados pela equipe de QA.
            \item Geração de documentação (\textit{Changelogs}).
            \item Atualização dos números de versão do sistema.
        \end{itemize}
    \end{itemize}
\end{frame}

% Slide 21 (Slide 11 do Fluxo)
\begin{frame}[fragile]{11. Criando a Release}
    A branch de release nasce a partir da \texttt{develop}.
    \vspace{0.3cm}
\begin{lstlisting}
# 1. Crie a release a partir da develop
git checkout -b release/1.2.0 develop

# 2. Ajustes de meta-dados (ex: mudanca de versao)
git commit -am "Bump version to 1.2.0"

# 3. Arrumando bugs de ultima hora durante homologacao
git commit -am "Corrige responsividade no mobile"
\end{lstlisting}
\end{frame}

% Slide 22 (Slide 12 do Fluxo)
\begin{frame}[fragile]{12. Finalizando a Release}
    A finalização da \texttt{release} é um \textbf{processo duplo}. Ela deve ir para produção (\texttt{main}) e voltar para a \texttt{develop} (para que as correções de bugs não sejam perdidas).
    \vspace{0.2cm}
\begin{lstlisting}
# 1. Merge na main e criacao da TAG
git checkout main
git merge --no-ff release/1.2.0
git tag -a v1.2.0 -m "Versao Estavel 1.2.0"

# 2. Merge de volta na develop
git checkout develop
git merge --no-ff release/1.2.0

# 3. Apagar a branch de release
git branch -d release/1.2.0
\end{lstlisting}
\end{frame}

% Slide 23 (Slide 13 do Fluxo)
\begin{frame}{13. Correções Emergenciais (\texttt{hotfix})}
    \begin{itemize}
        \item E se um bug catastrófico for descoberto na produção? O sistema de pagamentos caiu!
        \item Não podemos usar a \texttt{develop}, pois ela já possui recursos pela metade que não podem ir para a produção.
        \item O \textbf{\texttt{hotfix}} existe exatamente para isso: atuar de forma semelhante à release, mas iniciando \textbf{DIRETAMENTE} da \texttt{main} para isolar a correção.
    \end{itemize}
\end{frame}

% Slide 24 (Slide 14 do Fluxo)
\begin{frame}[fragile]{14. O Fluxo do Hotfix}
    Nasce na \texttt{main}, e retorna para \texttt{main} e \texttt{develop}.
    \vspace{0.2cm}
\begin{lstlisting}
# Nasce do estado exato da producao (main)
git checkout -b hotfix/1.2.1 main

# Commitando a correcao do bug
git commit -am "Corrige API de pagamentos (Emergencia)"

# Fechando e mandando para main (com nova TAG minor)
git checkout main
git merge --no-ff hotfix/1.2.1
git tag -a v1.2.1 -m "Hotfix Pagamento"

# Garantindo que a correcao exista na develop
git checkout develop
git merge --no-ff hotfix/1.2.1
\end{lstlisting}
\end{frame}

% Slide 25 (Slide 15 do Fluxo)
\begin{frame}{15. Versionamento com Tags}
    \begin{itemize}
        \item Tags funcionam como fotografias inalteráveis (snapshots) do projeto.
        \item \textbf{Automação:} Ferramentas de \textit{Continuous Deployment} (CD) modernas frequentemente monitoram o repositório. Se uma \texttt{Tag} chamada \texttt{v2.0.1} for criada, o sistema envia o código automaticamente para produção.
        \item \textbf{Semântica:} 
        \begin{itemize}
            \item Lançamentos regulares aumentam o dígito intermediário (\texttt{v1.3.0}).
            \item Correções de \texttt{hotfix} aumentam o último dígito (\texttt{v1.3.1}).
        \end{itemize}
    \end{itemize}
\end{frame}
